



for id = e1 to e2
do s

    IMC<e1>
    IMC<e2>
    Load_Mem R1, Result<e1>

startLabel:

    Compare_Less R2, R1 (U R2 je 1 ako je R2 manje od R1 i tad zavrsavamo)

    Jump startLabel




do s1 s2 s3 ... sn while (e)

continueLabel:
startLabel:
    IMC<s1>
    IMC<s2>
    IMC<s3>
    ...
    IMC<continue> (Jump startLabel)
    ...
    IMC<break> (Jump endLabel)
    ...
    IMC<sn>
    IMC<e>
    Load_Mem R1, Result<e>
    Compare_Equal R1, 0
    JumpIfZero(startLabel)
breakLabel:
endLabel:
